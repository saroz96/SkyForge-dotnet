# Step 1: Drop existing database
psql -h localhost -U postgres -c "SELECT pg_terminate_backend(pid) FROM pg_stat_activity WHERE datname = 'SkyForge' AND pid <> pg_backend_pid();"
psql -h localhost -U postgres -c "DROP DATABASE IF EXISTS \"SkyForge\";"

# Step 2: Create fresh database
psql -h localhost -U postgres -c "CREATE DATABASE \"SkyForge\";"

# Step 3: Full restore (includes schema and data)
pg_restore -h localhost -U postgres -d "SkyForge" --clean --if-exists "C:\Users\ASUS\Downloads\Ams_Backup.dump"
